# GitHub (for git operations and release downloads)
github.com
api.github.com

# npm registry (for potential package installations)
registry.npmjs.org

# Python packages (pip/uv installs)
pypi.org
files.pythonhosted.org

# Rust packages (cargo installs)
crates.io
static.crates.io
